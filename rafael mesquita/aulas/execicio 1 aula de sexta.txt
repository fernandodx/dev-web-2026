using Microsoft.AspNetCore.Mvc;

namespace ExercicioApi.Controllers
{
    public class AlunoRequest
    {
        public double Nota1 { get; set; }
        public double Nota2 { get; set; }
        public int Faltas { get; set; }
    }

    public class AlunoResponse
    {
        public double Media { get; set; }
        public string Situacao { get; set; } = string.Empty;
    }

    [ApiController]
    [Route("api/alunos")]
    public class AlunosController : ControllerBase
    {
        [HttpPost("avaliacao")]
        public IActionResult VerificarAvaliacao([FromBody] AlunoRequest request)
        {
            double media;
            string situacao;

            media = (request.Nota1 + request.Nota2) / 2.0;

            if (request.Faltas > 10)
            {
                situacao = "Reprovado por falta";
            }
            else if (media >= 7.0)
            {
                situacao = "Aprovado";
            }
            else
            {
                situacao = "Reprovado por nota";
            }

            AlunoResponse resposta = new AlunoResponse();
            resposta.Media = Math.Round(media, 2);
            resposta.Situacao = situacao;

            return Ok(resposta);
        }
    }
}

Web API (NumerosController.cs)



using Microsoft.AspNetCore.Mvc;

namespace ExercicioApi.Controllers
{
    public class InverterNumerosRequest
    {
        public Int32[] Numeros { get; set; } = new Int32[0];
    }

    public class InverterNumerosResponse
    {
        public Int32[] OrdemOriginal { get; set; } = new Int32[0];
        public Int32[] OrdemInversa { get; set; } = new Int32[0];
    }

    [ApiController]
    [Route("api/numeros")]
    public class NumerosController : ControllerBase
    {
        [HttpPost("inverter")]
        public IActionResult InverterVetor([FromBody] InverterNumerosRequest request)
        {
            Int32 tamanho = request.Numeros.Length;
            Int32[] original = new Int32[tamanho];
            Int32[] inverso = new Int32[tamanho];

            for (Int32 i = 0; i < tamanho; i++)
            {
                original[i] = request.Numeros[i];
            }

            Int32 posicaoInversa = 0;
            for (Int32 i = tamanho - 1; i >= 0; i--)
            {
                inverso[posicaoInversa] = request.Numeros[i];
                posicaoInversa++;
            }

            InverterNumerosResponse resposta = new InverterNumerosResponse();
            resposta.OrdemOriginal = original;
            resposta.OrdemInversa = inverso;

            return Ok(resposta);
        }
    }
}

Web API (PesquisaController.cs)




using Microsoft.AspNetCore.Mvc;

namespace ExercicioApi.Controllers
{
    public class PesquisaRequest
    {
        public Int32[] Numeros { get; set; } = new Int32[0];
        public Int32 NumeroPesquisa { get; set; }
    }

    public class PesquisaResponse
    {
        public string Status { get; set; } = string.Empty;
        public Int32 QuantidadeOcorrencias { get; set; }
        public Int32[] Posicoes { get; set; } = new Int32[0];
    }

    [ApiController]
    [Route("api/numeros")]
    public class PesquisaController : ControllerBase
    {
        [HttpPost("pesquisar")]
        public IActionResult PesquisarNumero([FromBody] PesquisaRequest request)
        {
            Int32 tamanho = request.Numeros.Length;
            Int32 ocorrencias = 0;

            for (Int32 i = 0; i < tamanho; i++)
            {
                if (request.Numeros[i] == request.NumeroPesquisa)
                {
                    ocorrencias = ocorrencias + 1;
                }
            }

            Int32[] posicoesEncontradas = new Int32[ocorrencias];
            Int32 indexPosicao = 0;

            for (Int32 i = 0; i < tamanho; i++)
            {
                if (request.Numeros[i] == request.NumeroPesquisa)
                {
                    posicoesEncontradas[indexPosicao] = i;
                    indexPosicao = indexPosicao + 1;
                }
            }

            PesquisaResponse resposta = new PesquisaResponse();

            if (ocorrencias > 0)
            {
                resposta.Status = "Número encontrado.";
                resposta.QuantidadeOcorrencias = ocorrencias;
                resposta.Posicoes = posicoesEncontradas;
            }
            else
            {
                resposta.Status = "Número não encontrado.";
                resposta.QuantidadeOcorrencias = 0;
                resposta.Posicoes = new Int32[0];
            }

            return Ok(resposta);
        }
    }
}

using Microsoft.AspNetCore.Mvc;

namespace ExercicioApi.Controllers
{
    public class ClassificarNumerosRequest
    {
        public Int32[] Numeros { get; set; } = new Int32[0];
    }

    public class ClassificarNumerosResponse
    {
        public Int32 QuantidadePositivos { get; set; }
        public Int32 QuantidadeNegativos { get; set; }
        public Int32 QuantidadeZeros { get; set; }
        public Int32 SomaPositivos { get; set; }
        public Int32 SomaNegativos { get; set; }
    }

    [ApiController]
    [Route("api/numeros")]
    public class ClassificarController : ControllerBase
    {
        [HttpPost("classificar")]
        public IActionResult ClassificarNumeros([FromBody] ClassificarNumerosRequest request)
        {
            Int32 tamanho = request.Numeros.Length;
            Int32 quantidadePositivos = 0;
            Int32 quantidadeNegativos = 0;
            Int32 quantidadeZeros = 0;
            Int32 somaPositivos = 0;
            Int32 somaNegativos = 0;

            for (Int32 i = 0; i < tamanho; i++)
            {
                if (request.Numeros[i] > 0)
                {
                    quantidadePositivos = quantidadePositivos + 1;
                    somaPositivos = somaPositivos + request.Numeros[i];
                }
                else if (request.Numeros[i] < 0)
                {
                    quantidadeNegativos = quantidadeNegativos + 1;
                    somaNegativos = somaNegativos + request.Numeros[i];
                }
                else
                {
                    quantidadeZeros = quantidadeZeros + 1;
                }
            }

            ClassificarNumerosResponse resposta = new ClassificarNumerosResponse();
            resposta.QuantidadePositivos = quantidadePositivos;
            resposta.QuantidadeNegativos = quantidadeNegativos;
            resposta.QuantidadeZeros = quantidadeZeros;
            resposta.SomaPositivos = somaPositivos;
            resposta.SomaNegativos = somaNegativos;

            return Ok(resposta);
        }
    }
}

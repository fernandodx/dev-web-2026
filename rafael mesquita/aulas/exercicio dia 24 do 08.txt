select timestampdiff(YEAR, Datanasc, CURDATE()) as "idade"
from funcionario
where Pnome like 'A%' AND Pnome like '%N';
   TT timestampdiff(YEAR, Datanasc, CURDATE()) as "idade"(~Pnome like 'A%' AND Pnome like '%N'(funcionario))
func <- (funcionario)
res <- ~Pnome like 'A%' AND Pnome like '%N'(func)
proj <- TT timestampdiff(YEAR, Datanasc, CURDATE()) as "idade" (res)

select max(Salario)
from funcionario join departamento on(Dnr = Dnumero)
where Dnome = 'Pesquisa';
   TT max(Salario)(~Dnome='Pesquisa'(funcionario |x|~Dnr=Dnumero departamento))
junc <- (funcionario |x|~Dnr=Dnumero departamento)
res <- ~Dnome='Pesquisa'(junc)
proj <- TT max(Salario) (res)

j)
select Pnome, Unome, Dlocal, Dnome, Horas
from funcionario
join departamento on(Dnr = Dnumero)
join localizacoes_dep on(Dnumero = localizacoes_dep.Dnumero)
join trabalha_em on(Cpf = Fcpf)
where Horas = (select max(Horas) from trabalha_em);
   TT Pnome, Unome, Dlocal, Dnome, Horas(~Horas=max(Horas)(funcionario |x|~Dnr=Dnumero departamento |x|~Dnumero=localizacoes_dep.Dnumero localizacoes_dep |x|~Cpf=Fcpf trabalha_em))
junc1 <- (funcionario |x|~Dnr=Dnumero departamento)
junc2 <- (junc1 |x|~Dnumero=localizacoes_dep.Dnumero localizacoes_dep)
junc3 <- (junc2 |x|~Cpf=Fcpf trabalha_em)
res <- ~Horas=max(Horas)(junc3)
proj <- TT Pnome, Unome, Dlocal, Dnome, Horas (res)

select Pnome, Unome
from funcionario
join trabalha_em on(Cpf = Fcpf)
join projeto on(Pnr = Projnumero)
where (Projlocal = 'São Paulo' AND Horas >= 10 AND Horas <= 20)
   OR (Projlocal = 'Rio de Janeiro' AND Horas >= 15);
   TT Pnome, Unome(~(Projlocal='São Paulo' AND Horas>=10 AND Horas<=20) OR (Projlocal='Rio de Janeiro' AND Horas>=15)(funcionario |x|~Cpf=Fcpf trabalha_em |x|~Pnr=Projnumero projeto))
junc1 <- (funcionario |x|~Cpf=Fcpf trabalha_em)
junc2 <- (junc1 |x|~Pnr=Projnumero projeto)
res <- ~(Projlocal='São Paulo' AND Horas>=10 AND Horas<=20) OR (Projlocal='Rio de Janeiro' AND Horas>=15)(junc2)
proj <- TT Pnome, Unome (res)
